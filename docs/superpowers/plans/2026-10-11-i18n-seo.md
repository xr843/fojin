# 多语言可收录（繁体 / 英文 SEO）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让首页、静态 shell、经典导读、专题、经文页的繁体（`?lang=zh-Hant`）与英文（`?lang=en`）版本能被搜索引擎单独收录。

**Architecture:** URL 不变，`?lang=` 区分语言。服务端三处出口（构建期 shell + nginx 选择、`seo.py` 经文页、sitemap）按语言输出 `<html lang>`、文案、自引用 canonical、hreflang；前端 Helmet 用同一套 URL 规则保持一致。辞典 / 人物 / 分享页不分语言。

**Tech Stack:** FastAPI + OpenCC（`opencc-python-reimplemented`，已在依赖里）；Vite 插件 + `opencc-js`（已在依赖里）；nginx `map`；React Helmet；pytest / vitest。

Spec：`docs/superpowers/specs/2026-10-11-i18n-seo-design.md`

## Global Constraints

- 语言值只认 `zh-Hant`、`en`；其他（含缺省、`zh`、未知值）一律视为简体 `zh`，canonical 为无参 URL。
- URL 拼法：`https://fojin.app{path}`（zh）、`https://fojin.app{path}?lang=zh-Hant`、`https://fojin.app{path}?lang=en`；不携带其他查询参数。
- hreflang 四条：`zh-Hans`→zh URL、`zh-Hant`、`en`、`x-default`→zh URL。
- `<html lang>`：`zh-Hans` / `zh-Hant` / `en`；`og:locale`：`zh_CN` / `zh_TW` / `en_US`。
- 简体页经名：「简体（繁體）」并列，两者相同只写一次；繁体页、英文页经名原样（库内即繁体）。
- 辞典 `/dict/*`、人物 `/persons/*`：只把 `<html lang="zh-Hans">` 改成 `<html lang="zh">`，不加 hreflang、不加语言版本。
- `/share/qa/*`、`/texts/:id/read`（noindex）：不分语言（read 页 canonical 不变）。
- 预渲染 Worker `workers/prerender/`：代码不动，只在 README 注明路由未生效。
- 新增断言必须先在未改代码上跑红（`verify-the-test-is-red-first`）。
- 提交信息用 conventional commits，**不加任何 Claude 署名 trailer**。
- 后端 lint：`ruff==0.9.7`；前端：`npm run lint`（0 warning）、`npm run i18n:check`、`npx tsc -b --noEmit`。

```python
# backend/app/services/seo_lang.py
"""Per-language URL / tag helpers for crawler-facing HTML.

FoJin serves three UI languages from the same path, distinguished only by
``?lang=``. Search engines index each variant separately only if every
variant self-canonicalises and all of them cross-reference each other via
hreflang — see docs/superpowers/specs/2026-10-11-i18n-seo-design.md.
"""

import logging
from typing import Literal

from fastapi import Request
from opencc import OpenCC

logger = logging.getLogger(__name__)

SeoLang = Literal["zh", "zh-Hant", "en"]

LANG_SHELL_SUFFIX: dict[str, str] = {"zh": "", "zh-Hant": ".zh-Hant", "en": ".en"}
_HTML_LANG = {"zh": "zh-Hans", "zh-Hant": "zh-Hant", "en": "en"}
_OG_LOCALE = {"zh": "zh_CN", "zh-Hant": "zh_TW", "en": "en_US"}
_t2s = OpenCC("t2s")


def resolve_lang(request: Request) -> SeoLang:
    value = request.query_params.get("lang", "")
    if value == "zh-Hant":
        return "zh-Hant"
    if value == "en":
        return "en"
    return "zh"


def lang_url(base: str, path: str, lang: SeoLang) -> str:
    url = f"{base}{path}"
    return url if lang == "zh" else f"{url}?lang={lang}"


def hreflang_tags(base: str, path: str) -> str:
    pairs = [
        ("zh-Hans", lang_url(base, path, "zh")),
        ("zh-Hant", lang_url(base, path, "zh-Hant")),
        ("en", lang_url(base, path, "en")),
        ("x-default", lang_url(base, path, "zh")),
    ]
    return "\n  ".join(f'<link rel="alternate" hreflang="{code}" href="{href}" />' for code, href in pairs)


def html_lang(lang: SeoLang) -> str:
    return _HTML_LANG[lang]


def og_locale(lang: SeoLang) -> str:
    return _OG_LOCALE[lang]


def localize_title(title: str, lang: SeoLang) -> str:
    if lang != "zh" or not title:
        return title
    try:
        simplified = _t2s.convert(title)
    except Exception as e:  # OpenCC failure must never break a page
        logger.warning("t2s conversion failed for %r: %s", title, e)
        return title
    return simplified if simplified == title else f"{simplified}（{title}）"
```

async def _fetch_index_html(suffix: str = "") -> str:
    now = time.monotonic()
    hit = _index_html_cache.get(suffix)
    if hit and now - hit[0] < _INDEX_HTML_CACHE_TTL:
        return hit[1]
    url = f"http://frontend/index{suffix}.html"
    async with httpx.AsyncClient(timeout=5.0) as client:
        resp = await client.get(url)
        if resp.status_code == 404 and suffix:
            resp = await client.get(_FRONTEND_INDEX_URL)
        resp.raise_for_status()
        html = resp.text
    _index_html_cache[suffix] = (now, html)

```ts
// frontend/src/seo/langUrl.ts
export type SeoLang = "zh" | "zh-Hant" | "en";
const ORIGIN = "https://fojin.app";

export function toSeoLang(lng: string): SeoLang {
  if (lng === "zh-Hant" || lng.startsWith("zh-Hant")) return "zh-Hant";
  if (lng === "en" || lng.startsWith("en-")) return "en";
  return "zh";
}

export function langUrl(path: string, lang: SeoLang): string {
  const url = `${ORIGIN}${path}`;
  return lang === "zh" ? url : `${url}?lang=${lang}`;
}

const OG: Record<SeoLang, string> = { zh: "zh_CN", "zh-Hant": "zh_TW", en: "en_US" };
export function ogLocaleFor(lang: SeoLang): string {
  return OG[lang];
}
```

```ts
// frontend/src/seo/syncLangParam.ts
import { toSeoLang } from "./langUrl";

/** 切换界面语言后把 ?lang= 写进地址栏，分享出去的链接才带着语言。 */
export function syncLangParam(lang: string, loc: Location = window.location, hist: History = window.history): void {
  const url = new URL(loc.href);
  const seo = toSeoLang(lang);
  if (seo === "zh") url.searchParams.delete("lang");
  else url.searchParams.set("lang", seo);
  const qs = url.searchParams.toString();
  hist.replaceState(hist.state ?? null, "", `${url.pathname}${qs ? `?${qs}` : ""}${url.hash}`);
}
```
